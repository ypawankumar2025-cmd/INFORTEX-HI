cmake_minimum_required(VERSION 3.20)
project(HumanIntelligence VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(LibArchive REQUIRED)
find_package(PostgreSQL QUIET)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
add_library(hi_core src/database.cpp src/provider.cpp src/research.cpp src/agents.cpp src/knowledge.cpp src/service.cpp)
target_include_directories(hi_core PUBLIC include ${ASIO_INCLUDE_DIR} ../vendor/Crow/include)
target_link_libraries(hi_core PUBLIC Threads::Threads OpenSSL::Crypto CURL::libcurl SQLite::SQLite3 nlohmann_json::nlohmann_json LibArchive::LibArchive)
target_compile_definitions(hi_core PUBLIC CROW_DISABLE_STATIC_DIR)
target_compile_options(hi_core PRIVATE -Wall -Wextra -Wpedantic)
if(PostgreSQL_FOUND)
 target_link_libraries(hi_core PUBLIC PostgreSQL::PostgreSQL)
 target_compile_definitions(hi_core PUBLIC HI_POSTGRES)
endif()
add_executable(hi_server src/main.cpp)
target_link_libraries(hi_server PRIVATE hi_core)
enable_testing()
add_executable(hi_tests tests/unit.cpp)
target_link_libraries(hi_tests PRIVATE hi_core)
add_test(NAME core COMMAND hi_tests ${CMAKE_CURRENT_SOURCE_DIR}/../database/schema.sql)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(hi_tests PRIVATE -UNDEBUG)
endif()
add_executable(hi_provider_transport_tests tests/provider_transport.cpp)
target_link_libraries(hi_provider_transport_tests PRIVATE hi_core)
